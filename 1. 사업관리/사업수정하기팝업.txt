import { useState, useEffect } from "react";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";

export default function ProjectEditModal({ project = {}, onClose, onSave }) {
  const [name, setName] = useState("");
  const [target, setTarget] = useState("");
  const [startDate, setStartDate] = useState("");
  const [endDate, setEndDate] = useState("");
  const [noticeUrl, setNoticeUrl] = useState("");
  const [organizer, setOrganizer] = useState("");

  useEffect(() => {
    if (project) {
      setName(project.name || "");
      setTarget(String(project.target || ""));
      setStartDate(project.startDate || "");
      setEndDate(project.endDate || "");
      setNoticeUrl(project.noticeUrl || "");
      setOrganizer(project.organizer || "");
    }
  }, [project]);

  const isValidUrl = (url) =>
    url === "" || url.startsWith("http://") || url.startsWith("https://");

  const handleUpdate = () => {
    if (!name || !target || !startDate || !endDate || !organizer) {
      alert("모든 항목을 입력해주세요.");
      return;
    }

    if (!isValidUrl(noticeUrl)) {
      alert("공고 URL은 http:// 또는 https:// 로 시작해야 합니다.");
      return;
    }

    const updated = {
      ...project,
      name,
      target: Number(target),
      startDate,
      endDate,
      noticeUrl,
      organizer,
    };

    onSave(updated);
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[460px]">
        <h2 className="text-lg font-semibold mb-4">✏️ 사업 정보 수정</h2>

        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium mb-1">사업명</label>
            <Input
              placeholder="클라우드 고도화 지원사업"
              value={name}
              onChange={(e) => setName(e.target.value)}
            />
          </div>

          <div>
            <label className="block text-sm font-medium mb-1">목표 업체 수</label>
            <Input
              type="number"
              placeholder="30"
              value={target}
              onChange={(e) => setTarget(e.target.value)}
            />
          </div>

          <div className="flex space-x-2">
            <div className="flex-1">
              <label className="block text-sm font-medium mb-1">시작일</label>
              <Input
                type="date"
                value={startDate}
                onChange={(e) => setStartDate(e.target.value)}
              />
            </div>
            <div className="flex-1">
              <label className="block text-sm font-medium mb-1">종료일</label>
              <Input
                type="date"
                value={endDate}
                onChange={(e) => setEndDate(e.target.value)}
              />
            </div>
          </div>

          <div>
            <label className="block text-sm font-medium mb-1">주최기관</label>
            <Input
              placeholder="예: 중소벤처기업부"
              value={organizer}
              onChange={(e) => setOrganizer(e.target.value)}
            />
          </div>

          <div>
            <label className="block text-sm font-medium mb-1">공고 URL</label>
            <Input
              placeholder="https://example.com/notice"
              value={noticeUrl}
              onChange={(e) => setNoticeUrl(e.target.value)}
            />
          </div>
        </div>

        <div className="flex justify-end space-x-2 mt-6">
          <Button variant="outline" onClick={onClose}>취소</Button>
          <Button onClick={handleUpdate}>저장</Button>
        </div>
      </div>
    </div>
  );
}
