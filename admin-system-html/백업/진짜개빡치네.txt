    async function fetchData(sheet, id = null, extraParams = {}) {
      try {
        let url = `${API_URL}?action=getData&sheet=${sheet}`;
        
        if (id) {
          url += `&id=${id}`;
        }
        
        // 추가 매개변수 적용
        for (const [key, value] of Object.entries(extraParams)) {
          url += `&${key}=${value}`;
        }
        
        // 콜백 이름 생성
        const callbackName = 'jsonpCallback_' + Math.round(Math.random() * 1000000);
        url += `&callback=${callbackName}`;
        
        // JSONP 호출
        return new Promise((resolve, reject) => {
          // 콜백 함수 등록
          window[callbackName] = function(response) {
            // 콜백 함수 제거
            delete window[callbackName];
            
            // 스크립트 태그 제거
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            
            resolve(response);
          };
          
          // 스크립트 태그 생성 및 추가
          const script = document.createElement('script');
          script.src = url;
          
          // 오류 처리
          script.onerror = function() {
            delete window[callbackName];
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            reject(new Error('네트워크 요청 실패'));
          };
          
          // 타임아웃 설정 (30초)
          const timeout = setTimeout(function() {
            delete window[callbackName];
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            reject(new Error('요청 시간 초과 (30초)'));
          }, 30000);
          
          // 원래 콜백 함수를 래핑하여 타임아웃 제거
          const originalCallback = window[callbackName];
          window[callbackName] = function(response) {
            clearTimeout(timeout);
            originalCallback(response);
          };
          
          // 스크립트 추가
          document.body.appendChild(script);
        });
      } catch (error) {
        console.error('API 호출 오류:', error);
        throw error;
      }
    }

    function getFilteredCompanies() {
      const searchTerm = document.getElementById('searchInput').value.trim().toLowerCase();
      if (!searchTerm) return companies;
      
      return companies.filter(c => 
        (c.comname && c.comname.toLowerCase().includes(searchTerm)) || 
        (c.businessNumber && c.businessNumber.toLowerCase().includes(searchTerm)) ||
        (c.representative && c.representative.toLowerCase().includes(searchTerm))
      );
    }
    
    function renderTable() {
      const tbody = document.getElementById('companyTableBody');
      const filtered = getFilteredCompanies();
      
      // 정렬 - 수동 정렬이 적용된 경우에만 정렬 실행
      let displayData = filtered;
      if (currentSortColumn !== 'id' || currentSortDirection !== 'asc') {
        displayData = filtered.sort((a, b) => {
          let valA = a[currentSortColumn] || '';
          let valB = b[currentSortColumn] || '';
          
          valA = String(valA).toLowerCase();
          valB = String(valB).toLowerCase();
          
          return (valA < valB ? -1 : valA > valB ? 1 : 0) * (currentSortDirection === 'asc' ? 1 : -1);
        });
      }
      
      // 초기 테이블 정렬 표시 업데이트
      if (currentSortColumn !== 'id') {
        document.querySelectorAll('th[data-sort]').forEach(th => {
          const column = th.getAttribute('data-sort');
          const sortIcon = th.querySelector('.sort-icon i');
          
          if (column === currentSortColumn) {
            sortIcon.className = `fas fa-sort-${currentSortDirection === 'asc' ? 'up' : 'down'}`;
            sortIcon.parentElement.classList.add('active');
          } else {
            sortIcon.className = 'fas fa-sort';
            sortIcon.parentElement.classList.remove('active');
          }
        });
      }
      
      // 페이징
      const pageData = displayData.slice((currentPage - 1) * itemsPerPage, currentPage * itemsPerPage);
      const fragment = document.createDocumentFragment();
      
      if (pageData.length === 0) {
        const tr = document.createElement('tr');
        tr.innerHTML = '<td colspan="8" class="text-center py-4">등록된 기업이 없습니다.</td>';
        fragment.appendChild(tr);
      } else {
        pageData.forEach(company => {
          const tr = document.createElement('tr');
          
          // 체크박스
          const checkboxCell = document.createElement('td');
          checkboxCell.className = "align-middle text-center";
          
          const checkboxContainer = document.createElement('div');
          checkboxContainer.className = "flex justify-center items-center";
          
          const checkbox = document.createElement('input');
          checkbox.type = 'checkbox';
          checkbox.className = 'company-checkbox';
          checkbox.value = company.id;
          checkbox.checked = selectedCompanyIds.includes(company.id);
          checkbox.addEventListener('change', () => {
            if (checkbox.checked) {
              if (!selectedCompanyIds.includes(company.id)) selectedCompanyIds.push(company.id);
            } else {
              selectedCompanyIds = selectedCompanyIds.filter(id => id !== company.id);
            }
            updateSelectAllCheckbox();
          });
          
          checkboxContainer.appendChild(checkbox);
          checkboxCell.appendChild(checkboxContainer);
          tr.appendChild(checkboxCell);
          tr.appendChild(createElement('td', company.comname)); // 기업명 (name → comname)
          tr.appendChild(createElement('td', company.businessNumber));
          tr.appendChild(createElement('td', company.representative));
          tr.appendChild(createElement('td', company.phone));
          tr.appendChild(createElement('td', company.email));
          tr.appendChild(createElement('td', company.registrationDate));
          
          // 수정 버튼
          const editBtn = document.createElement('button');
          editBtn.type = 'button';
          editBtn.className = 'btn btn-sm edit-btn whitespace-nowrap';
          editBtn.innerHTML = '<i class="fas fa-edit mr-1 flex-shrink-0"></i> 수정';
          editBtn.addEventListener('click', () => openModal('edit', company));
          
          const tdEdit = document.createElement('td');
          tdEdit.className = "text-center"; // 셀 자체도 중앙 정렬
          tdEdit.appendChild(editBtn);
          tr.appendChild(tdEdit);
          fragment.appendChild(tr);
        });
      }
      
      tbody.innerHTML = '';
      tbody.appendChild(fragment);
      
      // 페이지네이션 업데이트
      const totalPages = Math.ceil(filtered.length / itemsPerPage);
      const startItem = filtered.length === 0 ? 0 : (currentPage - 1) * itemsPerPage + 1;
      const endItem = Math.min(currentPage * itemsPerPage, filtered.length);
      
      document.getElementById('totalItems').textContent = filtered.length;

      document.getElementById('endItem').textContent = endItem;

      // 전체 선택 체크박스 상태 업데이트
      updateSelectAllCheckbox();
    }

    function updateSelectAllCheckbox() {
      const checkboxes = document.querySelectorAll('.company-checkbox');
      const selectAllCheckbox = document.getElementById('selectAllCheckbox');
      
      selectAllCheckbox.disabled = checkboxes.length === 0;
      selectAllCheckbox.checked = checkboxes.length > 0 && Array.from(checkboxes).every(chk => chk.checked);
      
      // 삭제 버튼 활성화/비활성화
      const deleteButton = document.getElementById('deleteButton');
      const anyChecked = checkboxes.length > 0 && Array.from(checkboxes).some(chk => chk.checked);
      
      if (anyChecked) {
        deleteButton.disabled = false;
        deleteButton.classList.remove('opacity-50', 'cursor-not-allowed');
      } else {
        deleteButton.disabled = true;
        deleteButton.classList.add('opacity-50', 'cursor-not-allowed');
      }
    }

    // 파일 업로드 설정
    function setupFileUpload() {
      const fileInput = document.getElementById('companyFileInput');
      const fileNameDisplay = document.getElementById('selectedFileName');
      const dragDropArea = document.getElementById('dragDropArea');
      
      if (!fileInput || !fileNameDisplay || !dragDropArea) return;
      
      // 기존 이벤트 리스너 제거
      const newDragDropArea = dragDropArea.cloneNode(true);
      dragDropArea.parentNode.replaceChild(newDragDropArea, dragDropArea);
      
      // 새로운 파일 입력 참조 가져오기 (중요: cloneNode 후에는 새로운 참조를 가져와야 함)
      const newFileInput = newDragDropArea.querySelector('input[type="file"]');
      if (!newFileInput) {
        console.error('파일 입력 요소를 찾을 수 없습니다.');
        return;
      }
      
      // 파일 선택 이벤트
      newFileInput.addEventListener('change', function(e) {
        e.stopPropagation();
        const selectedFiles = this.files;
        updateSelectedFileInfo(selectedFiles);
        console.log('파일이 선택되었습니다:', this.files[0]?.name);
        
        // 선택된 파일이 있으면 즉시 내용 검증
        if (selectedFiles && selectedFiles.length > 0) {
          validateFileContent(selectedFiles[0]);
        }
      });
      
      // 드래그 앤 드롭 이벤트
      ['dragenter', 'dragover', 'dragleave', 'drop'].forEach(eventName => {
        newDragDropArea.addEventListener(eventName, preventDefaults, false);
      });
      
      function preventDefaults(e) {
        e.preventDefault();
        e.stopPropagation();
      }
      
      // 드래그 효과
      ['dragenter', 'dragover'].forEach(eventName => {
        newDragDropArea.addEventListener(eventName, highlight, false);
      });
      
      ['dragleave', 'drop'].forEach(eventName => {
        newDragDropArea.addEventListener(eventName, unhighlight, false);
      });
      
      function highlight() {
        newDragDropArea.classList.add('border-blue-500');
        newDragDropArea.classList.add('bg-blue-50');
      }
      
      function unhighlight() {
        newDragDropArea.classList.remove('border-blue-500');
        newDragDropArea.classList.remove('bg-blue-50');
      }
      
      // 파일 드롭 처리
      newDragDropArea.addEventListener('drop', handleDrop, false);
      
      function handleDrop(e) {
        const dt = e.dataTransfer;
        const files = dt.files;
        updateSelectedFileInfo(files);
        
        // 직접 DOM에서 파일 입력 필드 참조 다시 가져오기
        const fileInput = document.querySelector('#addModal #companyFileInput');
        if (fileInput && files.length > 0) {
          // 일부 브라우저에서는 files 속성에 직접 할당할 수 없으므로 DataTransfer 객체 사용
          try {
            // 데이터 전송 객체 생성
            const dataTransfer = new DataTransfer();
            // 첫 번째 파일만 추가
            dataTransfer.items.add(files[0]);
            // 파일 목록 설정
            fileInput.files = dataTransfer.files;
            console.log('드래그 앤 드롭으로 파일이 설정되었습니다:', fileInput.files[0]?.name);
            
            // 드롭된 파일 내용 즉시 검증
            validateFileContent(files[0]);
          } catch (error) {
            console.error('파일 설정 중 오류 발생:', error);
            // 오류 발생 시 사용자에게 알림
            alert('파일을 끌어다 놓는 기능이 지원되지 않습니다. 파일 선택 버튼을 사용해 주세요.');
          }
        }
      }
    }
    
    // 파일 정보 업데이트
    function updateSelectedFileInfo(files) {
      const fileNameDisplay = document.getElementById('selectedFileName');
      if (!fileNameDisplay) return;
      
      if (files.length > 0) {
        const file = files[0];
        
        // 파일 형식 확인
        const fileExtension = file.name.split('.').pop().toLowerCase();
        if (fileExtension !== 'csv') {
          fileNameDisplay.innerHTML = `<span class="text-red-500">${file.name} (지원되지 않는 파일 형식)</span>`;
        } else {
          fileNameDisplay.innerHTML = `<span class="text-green-500">${file.name} (${formatFileSize(file.size)})</span>`;
        }
      } else {
        fileNameDisplay.textContent = '선택된 파일 없음';
      }
    }
    
    // 파일 크기 포맷팅
    function formatFileSize(bytes) {
      if (bytes === 0) return '0 Bytes';
      const k = 1024;
      const sizes = ['Bytes', 'KB', 'MB', 'GB'];
      const i = Math.floor(Math.log(bytes) / Math.log(k));
      return parseFloat((bytes / Math.pow(k, i)).toFixed(2)) + ' ' + sizes[i];
    }

    // 파일 업로드 처리 - fetch API와 POST 방식 사용
    function processFileUpload(formData, resolve, reject) {
      const file = formData.get('file');
      const id = formData.get('id');
      
      console.log('📄 파일 업로드 처리 시작 - 파일명:', file.name);
      
      // FileReader를 사용하여 파일을 텍스트로 읽기
      const reader = new FileReader();
      
      reader.onload = function(e) {
        try {
          // 디버깅: 파일 내용 확인
          const content = e.target.result;
          console.log('📑 파일 내용 (처음 200자):', content.substring(0, 200));
          
          // 개선된 CSV 파싱 로직
          const rows = CSVToArray(content);
          console.log('📋 파싱된 CSV 행 개수:', rows.length);
          
          if (rows.length > 0) {
            const headers = rows[0];
            console.log('📋 파일 헤더:', headers);
            
            // 필수 컬럼 검사
            const requiredColumns = ['comname', 'businessNumber'];
            const missingColumns = [];
            
            requiredColumns.forEach(column => {
              if (headers.indexOf(column) === -1) {
                missingColumns.push(column);
              }
            });
            
            if (missingColumns.length > 0) {
              console.error('❌ 필수 컬럼 누락:', missingColumns.join(', '));
              reject(`파일 형식 오류: 필수 컬럼(${missingColumns.join(', ')})이 누락되었습니다.`);
              return;
            }
            
            // 나머지 파일 검증 로직 (필요한 경우 유지)
            // ...
          }
          
          // 파일 내용을 Base64로 인코딩
          let fileContent;
          try {
            fileContent = btoa(unescape(encodeURIComponent(content)));
          } catch (encodeError) {
            console.error('❌ 인코딩 오류:', encodeError);
            fileContent = btoa(content); // 폴백 방식
          }
          
          // POST 요청 데이터 준비
          const postData = {
            action: 'uploadFile',
            sheet: '기업정보',  // 항상 기업정보 시트로 명시적 지정
            id: id || null,
            file: {
              name: file.name,
              type: file.type,
              content: fileContent
            }
          };
          
          console.log('📤 API 요청 준비 완료 - JSONP 방식으로 전송');
          
          // JSONP 요청 생성 (롤백)
          const callbackName = 'jsonpCallback_' + Math.round(Math.random() * 1000000);
          const timestamp = new Date().getTime();
          const encodedData = encodeURIComponent(JSON.stringify(postData));
          const script = document.createElement('script');
          script.src = `${API_URL}?callback=${callbackName}&cache_buster=${timestamp}&data=${encodedData}`;
          
          console.log('🔗 API URL 생성됨 (길이)', script.src.length);
          
          // 콜백 함수 등록
          window[callbackName] = function(response) {
            // 콜백 함수 제거
            delete window[callbackName];
            
            // 스크립트 태그 제거
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            
            console.log('📥 API 응답 수신:', JSON.stringify(response).substring(0, 100) + '...');
            console.log('📥 API 응답 전체:', response); // 전체 응답 로깅
            
            if (response && response.success) {
              console.log('✅ API 요청 성공');
              // 추가된 항목 수 로깅
              if (response.data && response.data.companies) {
                console.log('📈 추가된 항목 수:', response.data.companies.length);
              }
              resolve(response);
            } else {
              console.error('❌ API 응답 오류:', response);
              reject(response && response.error ? response.error : '업로드 실패');
            }
          };
          
          // 오류 처리
          script.onerror = function() {
            // 콜백 함수 제거
            delete window[callbackName];
            
            // 스크립트 태그 제거
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            
            console.error('❌ 네트워크 오류 발생');
            reject('네트워크 오류가 발생했습니다.');
          };
          
          // 타임아웃 설정
          const timeout = setTimeout(function() {
            // 콜백 함수 제거
            delete window[callbackName];
            
            // 스크립트 태그 제거
            if (script.parentNode) {
              script.parentNode.removeChild(script);
            }
            
            console.error('⏱️ 요청 시간 초과');
            reject('요청 시간이 초과되었습니다. (30초)');
          }, 30000);
          
          // 원래 콜백 함수를 래핑하여 타임아웃 제거
          const originalCallback = window[callbackName];
          window[callbackName] = function(response) {
            clearTimeout(timeout);
            originalCallback(response);
          };
          
          // 스크립트 추가 (body에 추가)
          document.body.appendChild(script);
          console.log('📡 JSONP 요청이 전송되었습니다.');
        } catch (error) {
          console.error('❌ 파일 처리 오류:', error);
          reject('파일 처리 중 오류가 발생했습니다: ' + error.message);
        }
      }; // reader.onload 함수 끝
      
      reader.onerror = function(error) {
        console.error('❌ 파일 읽기 오류:', error);
        reject('파일을 읽는 중 오류가 발생했습니다.');
      };
      
      // CSV 파일 읽기
      reader.readAsText(file, 'UTF-8');
    }
    
    // CSV 문자열을 2차원 배열로 변환하는 함수 추가
    function CSVToArray(csvString, delimiter = ',') {
      // 빈 문자열이면 빈 배열 반환
      if (csvString.length === 0) return [[]];
      
      const rows = [];      // 전체 행 배열
      let row = [];         // 현재 행 배열
      let i = 0;            // 현재 문자 인덱스
      let char;             // 현재 문자
      let nextChar;         // 다음 문자
      let insideQuotes = false; // 따옴표 내부 여부
      let cellContent = ''; // 현재 셀 내용
      
      // 문자열의 각 문자를 순회
      while (i < csvString.length) {
        char = csvString[i];
        nextChar = i < csvString.length - 1 ? csvString[i + 1] : '';
        
        // 따옴표 처리
        if (char === '"') {
          // 따옴표 내부에 있고, 다음 문자도 따옴표인 경우 (이스케이프된 따옴표)
          if (insideQuotes && nextChar === '"') {
            cellContent += '"';
            i += 2; // 두 개의 따옴표를 하나로 처리
            continue;
          }
          
          // 따옴표 상태 토글
          insideQuotes = !insideQuotes;
          i++;
          continue;
        }
        
        // 구분자 처리 (따옴표 외부에서만)
        if (char === delimiter && !insideQuotes) {
          row.push(cellContent.trim());
          cellContent = '';
          i++;
          continue;
        }
        
        // 줄바꿈 처리 (따옴표 외부에서만)
        if ((char === '\r' || char === '\n') && !insideQuotes) {
          // 이전 문자가 '\r'이고 현재 문자가 '\n'인 경우 (CRLF)
          if (csvString[i - 1] === '\r' && char === '\n') {
            i++;
            continue;
          }
          
          // 빈 행이 아닌 경우에만 행 추가
          if (row.length > 0 || cellContent.length > 0) {
            row.push(cellContent.trim());
            rows.push(row);
            row = [];
            cellContent = '';
          }
          
          i++;
          continue;
        }
        
        // 일반 문자 추가
        cellContent += char;
        i++;
      }
      
      // 마지막 행 처리
      if (row.length > 0 || cellContent.length > 0) {
        row.push(cellContent.trim());
        rows.push(row);
      }
      
      return rows;
    }

    // 기업 목록 다운로드 - CSV 형식으로 제한
    function downloadCompanyList() {
      console.log('기업 목록 다운로드');
      
      // 다운로드 버튼 상태 변경
      const button = document.getElementById('excelButton');
      const originalText = button.innerHTML;
      button.innerHTML = '<i class="fas fa-spinner fa-spin mr-2"></i>다운로드 중...';
      button.disabled = true;
      
      try {
        // 데이터가 없는 경우 안내
        if (companies.length === 0) {
          alert('다운로드할 기업 정보가 없습니다.');
          return;
        }
        
        // BOM 추가 (UTF-8)
        const BOM = new Uint8Array([0xEF, 0xBB, 0xBF]);
        
        // 고정된 컬럼 형식으로 헤더 정의 (항상 동일한 순서 유지)
        const csvColumns = [
          { key: 'comname', header: '기업명' },
          { key: 'businessNumber', header: '사업자등록번호' },
          { key: 'representative', header: '대표자' },
          { key: 'phone', header: '전화번호' },
          { key: 'email', header: '이메일' },
        ];
        
        // 헤더 행 생성
        let csvContent = csvColumns.map(col => col.header).join(',') + '\n';
        
        // 각 기업 데이터 행 생성
        companies.forEach(company => {
          // 각 컬럼을 정의된 순서대로 추출
          const rowData = csvColumns.map(col => {
            // 데이터가 없으면 빈 문자열 반환
            const value = company[col.key] || '';
            return escapeCsv(value);
          });
          
          // 행을 CSV 라인으로 추가
          csvContent += rowData.join(',') + '\n';
        });
        
        // Blob에 BOM과 CSV 내용 추가
        const blob = new Blob([BOM, csvContent], { type: 'text/csv;charset=utf-8' });
        const url = URL.createObjectURL(blob);
        
        // 다운로드 링크 생성 및 클릭
        const link = document.createElement('a');
        link.setAttribute('href', url);
        link.setAttribute('download', '기업목록.csv');
        link.style.visibility = 'hidden';
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
        
        console.log('✅ CSV 다운로드 완료');
        alert('기업 목록이 다운로드되었습니다.');
      } catch (error) {
        console.error('다운로드 오류:', error);
        alert('다운로드 중 오류가 발생했습니다.');
      } finally {
        // 버튼 상태 복원
        button.innerHTML = originalText;
        button.disabled = false;
      }
      
      // CSV 문자열 이스케이프 함수
      function escapeCsv(text) {
        if (!text) return '';
        text = String(text).trim().replace(/"/g, '""'); // 앞뒤 공백 제거 및 따옴표 이스케이프
        if (text.includes(',') || text.includes('\n') || text.includes('"')) {
          return `"${text}"`; 
        }
        return text;
      }
    }

          console.log('📋 헤더:', headers);
          
          // 필수 컬럼 확인
          const requiredColumns = ['comname', 'businessNumber', 'representative', 'phone', 'email'];
          const missingColumns = [];
          
          requiredColumns.forEach(column => {
            if (headers.indexOf(column) === -1) {
              missingColumns.push(column);
            }
          });
          
          if (missingColumns.length > 0) {
            alert(`필수 컬럼이 누락되었습니다: ${missingColumns.join(', ')}`);
            clearFileInput();
            return;
          }
          
          // 데이터 행 확인
          if (rows.length < 2) {
            alert('파일에 기업 정보가 없습니다. 최소 한 개 이상의 데이터 행이 필요합니다.');
            clearFileInput();
            return;
          }
          
          // 데이터 행의 내용 확인
          const companyNameIndex = headers.indexOf('comname');
          const businessNumberIndex = headers.indexOf('businessNumber');
          let emptyDataCount = 0;
          let existingCompanies = 0;
          let newCompanies = 0;
          
          // 사업자등록번호 정규화 함수 (하이픈 등 특수문자 제거)
          function normalizeBusinessNumber(number) {
            if (!number) return '';
            
            // 숫자가 아닌 모든 문자(하이픈, 공백 등) 제거
            const normalized = String(number).replace(/[^0-9]/g, '');
            
            // 숫자가 하나도 없으면 빈 문자열 반환
            return normalized;
          }
          
          // 기존 사업자등록번호 목록 생성 (중복 확인용)
          // ID와 사업자등록번호를 함께 저장해 로깅 목적으로 사용
          const existingBusinessMap = {};
          
          companies.forEach(company => {
            if (!company) return;
            
            // 숫자만 추출하여 정규화
            const normalized = normalizeBusinessNumber(company.businessNumber || '');
            
            // 정규화된 사업자등록번호가 유효한 경우만 맵에 추가
            if (normalized && normalized.length > 0) {
              existingBusinessMap[normalized] = {
                id: company.id,
                name: company.comname || '',
                original: company.businessNumber || ''
              };
            }
          });
          
          // 사업자등록번호 배열 (유효한 것만)
          const existingBusinessNumbers = Object.keys(existingBusinessMap);
          
          console.log('기존 사업자등록번호 수:', existingBusinessNumbers.length);
          
          // 파일 내 유효한 데이터 행 배열
          const validRows = [];
          const duplicateDetails = []; // 중복 상세 정보
          
          for (let i = 1; i < rows.length; i++) {
            const row = rows[i];
            // 빈 행 건너뛰기
            if (row.every(cell => cell.trim() === '')) {
              emptyDataCount++;
              continue;
            }
            
            // 필수 값 확인
            if (!row[companyNameIndex] || row[companyNameIndex].trim() === '' || 
                !row[businessNumberIndex] || row[businessNumberIndex].trim() === '') {
              alert(`${i}번째 행에 comname 또는 businessNumber가 누락되었습니다.`);
              return; // 입력은 유지
            }
            
            // 현재 행 데이터
            const companyName = row[companyNameIndex].trim();
            const originalNum = row[businessNumberIndex].trim();
            const businessNumber = normalizeBusinessNumber(originalNum);
            
            // 유효한 행 추가
            validRows.push({
              index: i,
              name: companyName,
              businessNumber: businessNumber,
              originalNum: originalNum,
              row: row
            });
            
            // 사업자등록번호가 숫자가 아닌 경우 경고
            if (!businessNumber) {
              console.warn(`⚠️ ${i}행의 사업자등록번호가 유효하지 않습니다:`, originalNum);
            }
            
            // 기존 데이터인지 확인 (정규화된 번호로 비교)
            if (businessNumber && existingBusinessNumbers.includes(businessNumber)) {
              existingCompanies++;
              
              // 중복 상세 정보 저장
              const existingInfo = existingBusinessMap[businessNumber];
              duplicateDetails.push({
                rowIndex: i,
                newName: companyName,
                newNumber: originalNum,
                existingId: existingInfo.id,
                existingName: existingInfo.name,
                existingNumber: existingInfo.original
              });
              
              console.log(`✓ ${i}행: 기존 기업 "${existingInfo.name}" (${existingInfo.original})와 일치`);
            } else {
              newCompanies++;
              console.log(`+ ${i}행: 신규 기업 - ${companyName} (${originalNum})`);
            }
          }
          
          console.log('-------------- 검증 결과 --------------');
          console.log(`총 ${validRows.length}개 기업 중 신규: ${newCompanies}개, 기존: ${existingCompanies}개`);
          
          if (duplicateDetails.length > 0) {
            console.log('---------- 중복 기업 상세 정보 ----------');
            duplicateDetails.forEach((detail, idx) => {
              console.log(`${idx+1}. ${detail.rowIndex}행: "${detail.newName}" (${detail.newNumber}) ↔ 기존: "${detail.existingName}" (${detail.existingNumber})`);
            });
          }
          
          // 모든 행이 비어있는 경우
          if (validRows.length === 0) {
            alert('파일에 유효한 기업 정보가 없습니다.');
            clearFileInput();
            return;
          }
          
          // 중복 데이터만 있는 경우 경고
          if (newCompanies === 0 && existingCompanies > 0) {
            if (confirm('파일의 모든 기업이 이미 등록되어 있습니다. 계속 진행하시겠습니까?')) {
              // 사용자가 계속 진행하기로 선택
            } else {
              clearFileInput();
              return;
            }
          }
          
          // 검증 통과 시 표시 (기존/신규 구분)
          let fileInfoHTML = `<span class="text-green-500">${file.name} (${formatFileSize(file.size)})</span><br>` +
            `<span class="text-sm">총 ${validRows.length}개 기업 중 ` + 
            `<span class="text-blue-500 font-bold">${newCompanies}개 신규</span>`;
          
          // 중복 기업이 있는 경우 상세 정보 표시 버튼 추가
          if (existingCompanies > 0) {
            fileInfoHTML += `, <span class="text-gray-500">${existingCompanies}개 중복</span>`;
            fileInfoHTML += ` <button id="showDuplicateDetails" type="button" class="text-xs text-blue-600 underline">상세보기</button>`;
          }
          
          fileInfoHTML += `</span>`;
          
          // 중복 상세 정보 목록 (기본적으로 숨김)
          if (existingCompanies > 0) {
            fileInfoHTML += `<div id="duplicateDetailsList" class="mt-2 bg-gray-50 p-2 text-xs rounded border border-gray-200 hidden">`;
            fileInfoHTML += `<div class="font-bold mb-1">중복된 기업 목록:</div>`;
            fileInfoHTML += `<ul class="list-disc pl-5 space-y-1 max-h-32 overflow-y-auto">`;
            
            duplicateDetails.forEach(detail => {
              fileInfoHTML += `<li>${detail.rowIndex}행: <span class="font-medium">${detail.newName}</span> (${detail.newNumber}) - 기존: ${detail.existingName}</li>`;
            });
            
            fileInfoHTML += `</ul></div>`;
          }
          
          document.getElementById('selectedFileName').innerHTML = fileInfoHTML;
          
          // 상세보기 버튼 이벤트 리스너 추가
          if (existingCompanies > 0) {
            setTimeout(() => {
              const detailsButton = document.getElementById('showDuplicateDetails');
              const detailsList = document.getElementById('duplicateDetailsList');
              
              if (detailsButton && detailsList) {
                detailsButton.addEventListener('click', function() {
                  detailsList.classList.toggle('hidden');
                  this.textContent = detailsList.classList.contains('hidden') ? '상세보기' : '접기';
                });
              }
            }, 100);
          }
          
          console.log(`✅ 파일 검증 완료: 총 ${validRows.length}개 (신규 ${newCompanies}, 기존 ${existingCompanies})`);
        } catch (error) {
          console.error('파일 검증 중 오류:', error);
          alert('파일 내용을 확인하는 중 오류가 발생했습니다: ' + error.message);
          clearFileInput();
        }
      };
      
      reader.onerror = function() {
        console.error('파일 읽기 오류');
        alert('파일을 읽는 중 오류가 발생했습니다.');
        clearFileInput();
      };
      
      // 파일 읽기 시작
      reader.readAsText(file, 'UTF-8');
    }
    
    // 파일 입력 초기화 함수
    function clearFileInput() {
      const fileInput = document.querySelector('#addModal #companyFileInput');
      if (fileInput) {
        fileInput.value = '';
        document.getElementById('selectedFileName').textContent = '선택된 파일 없음';
      }
    }
  </script>
