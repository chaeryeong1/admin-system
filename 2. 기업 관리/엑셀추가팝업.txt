import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";

export default function CompanyUploadModal({ onClose, onUpload }) {
  const handleUpload = (e) => {
    const file = e.target.files?.[0];
    if (file) {
      onUpload(file);
    }
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[420px]">
        <h2 className="text-lg font-semibold mb-4">📎 기업 리스트 엑셀 추가</h2>
        <p className="text-sm text-gray-600 mb-4">
          기업 리스트를 엑셀 파일로 일괄 등록할 수 있어요.<br />
          아래 버튼을 눌러 엑셀 파일을 업로드 해주세요.
        </p>

        <div className="space-y-4">
          <Input type="file" accept=".xlsx, .xls" onChange={handleUpload} />
          <div className="text-sm text-gray-500">
            📄 예시 파일이 필요하신가요? <button className="text-blue-600 underline" onClick={() => alert("샘플 다운로드")}>샘플 다운로드</button>
          </div>
        </div>

        <div className="flex justify-end space-x-2 mt-6">
          <Button variant="outline" onClick={onClose}>닫기</Button>
        </div>
      </div>
    </div>
  );
}
