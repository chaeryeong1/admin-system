import { useEffect } from "react";
import { Button } from "@/components/ui/button";

export default function CompanyDeleteModal({ companyName = "주식회사 이엑스", ceo = "홍길동", bizNo = "123-45-67890", email = "ex@...", phone = "010-1234-5678", regDate = "2025-04-11", onConfirm, onCancel }) {
  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === "Escape") onCancel();
    };
    document.addEventListener("keydown", handleEsc);
    return () => document.removeEventListener("keydown", handleEsc);
  }, [onCancel]);

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[400px] transition-transform duration-200 scale-100">
        <h2 className="text-lg font-semibold mb-4">⚠️ '{companyName}' 기업 삭제 확인</h2>
        <div className="text-sm text-gray-700 mb-4 space-y-1">
          <p><strong>기업명:</strong> {companyName}</p>
          <p><strong>대표자:</strong> {ceo}</p>
          <p><strong>사업자등록번호:</strong> {bizNo}</p>
          <p><strong>이메일:</strong> {email}</p>
          <p><strong>연락처:</strong> {phone}</p>
          <p><strong>등록일:</strong> {regDate}</p>
        </div>
        <p className="text-sm text-red-600 mb-6">
          <p>삭제 후에는 복구할 수 없습니다.</p> 정말 삭제하시겠습니까?
        </p>
        <div className="flex justify-end space-x-2">
          <Button variant="outline" autoFocus onClick={onCancel}>취소</Button>
          <Button variant="destructive" onClick={onConfirm}>삭제</Button>
        </div>
      </div>
    </div>
  );
}
