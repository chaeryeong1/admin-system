import { useEffect, useState } from "react";
import { Input } from "@/components/ui/input";
import { Button } from "@/components/ui/button";

export default function CompanyEditModal({ company, onSave, onClose }) {
  const [name, setName] = useState("");
  const [bizNo, setBizNo] = useState("");
  const [ceo, setCeo] = useState("");
  const [email, setEmail] = useState("");
  const [phone, setPhone] = useState("");

  useEffect(() => {
    if (company) {
      setName(company.name || "");
      setBizNo(company.bizNo || "");
      setCeo(company.ceo || "");
      setEmail(company.email || "");
      setPhone(company.phone || "");
    }
  }, [company]);

  const handleSave = () => {
    if (!name || !bizNo || !ceo) {
      alert("필수 항목을 모두 입력해주세요.");
      return;
    }
    onSave({ name, bizNo, ceo, email, phone });
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[460px]">
        <h2 className="text-lg font-semibold mb-4 flex items-center space-x-2">
          <span>✏️</span>
          <span>기업 정보 수정</span>
        </h2>

        <div className="space-y-4">
          <div>
            <label className="block text-sm font-medium mb-1">기업명</label>
            <Input value={name} onChange={(e) => setName(e.target.value)} placeholder="주식회사 이엑스" />
          </div>
          <div>
            <label className="block text-sm font-medium mb-1">사업자등록번호</label>
            <Input value={bizNo} onChange={(e) => setBizNo(e.target.value)} placeholder="123-45-67890" />
          </div>
          <div>
            <label className="block text-sm font-medium mb-1">대표자</label>
            <Input value={ceo} onChange={(e) => setCeo(e.target.value)} placeholder="홍길동" />
          </div>
          <div>
            <label className="block text-sm font-medium mb-1">이메일</label>
            <Input type="email" value={email} onChange={(e) => setEmail(e.target.value)} placeholder="ex@example.com" />
          </div>
          <div>
            <label className="block text-sm font-medium mb-1">연락처</label>
            <Input value={phone} onChange={(e) => setPhone(e.target.value)} placeholder="010-1234-5678" />
          </div>
        </div>

        <div className="flex justify-end space-x-2 mt-6">
          <Button variant="outline" onClick={onClose}>취소</Button>
          <Button onClick={handleSave}>저장</Button>
        </div>
      </div>
    </div>
  );
}
