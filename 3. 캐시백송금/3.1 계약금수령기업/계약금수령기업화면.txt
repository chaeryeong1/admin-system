import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Table, TableHeader, TableRow, TableHead, TableBody, TableCell } from "@/components/ui/table";

const initialCompanies = [
  {
    id: 1,
    name: "주식회사 이엑스",
    bizNo: "123-45-67890",
    ceo: "홍길동",
    amount: 2000000,
    date: "2025-04-11",
  },
];

export default function CompanyPayoutList() {
  const [companies, setCompanies] = useState(initialCompanies);
  const [filter, setFilter] = useState("");
  const [selectedId, setSelectedId] = useState(null);

  const filtered = companies.filter((c) => c.name.includes(filter));

  return (
    <div className="p-6 space-y-4">
      <h2 className="text-xl font-bold pb-2 border-b">수령 기업</h2>

      <div className="flex gap-2 items-center">
        <select className="border rounded px-3 py-2 text-sm">
          <option>2025 클라우드 보급사업</option>
        </select>
        <Input
          placeholder="기업명 검색"
          value={filter}
          onChange={(e) => setFilter(e.target.value)}
          className="w-60"
        />
        <Button variant="outline">조회</Button>
        <div className="flex-1 text-right">
          <Button>기업 추가</Button>
        </div>
      </div>

      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>선택</TableHead>
            <TableHead>기업명</TableHead>
            <TableHead>사업자등록번호</TableHead>
            <TableHead>대표자</TableHead>
            <TableHead>수령액</TableHead>
            <TableHead>처리일</TableHead>
          </TableRow>
        </TableHeader>
        <TableBody>
          {filtered.map((c) => (
            <TableRow key={c.id}>
              <TableCell>
                <label className="inline-flex items-center">
                  <input
                    type="radio"
                    name="select"
                    className="w-4 h-4 text-blue-600 bg-gray-100 border-gray-300 focus:ring-blue-500 focus:ring-2 rounded-full"
                    checked={selectedId === c.id}
                    onChange={() => setSelectedId(c.id)}
                  />
                </label>
              </TableCell>
              <TableCell>
                <a href="#" className="text-blue-600 underline text-sm">
                  {c.name}
                </a>
              </TableCell>
              <TableCell>{c.bizNo}</TableCell>
              <TableCell>{c.ceo}</TableCell>
              <TableCell>{c.amount.toLocaleString()}원</TableCell>
              <TableCell>{c.date}</TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>

      <div className="flex justify-end gap-2">
        <Button variant="secondary" disabled>수정</Button>
        <Button variant="destructive" disabled>삭제</Button>
      </div>
    </div>
  );
}
