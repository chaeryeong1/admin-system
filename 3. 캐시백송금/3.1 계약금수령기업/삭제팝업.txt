import { useEffect } from "react";
import { Button } from "@/components/ui/button";
import { Trash2 } from "lucide-react";

export default function CompanyDeleteConfirmModal({ onConfirm, onClose, companyName }) {
  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", handleEsc);
    return () => document.removeEventListener("keydown", handleEsc);
  }, [onClose]);

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[420px]">
        <h2 className="text-lg font-semibold mb-4 flex items-center space-x-2">
          <Trash2 size={18} className="text-red-600" />
          <span>계약금 수령 기업 삭제</span>
        </h2>

        <p className="text-sm text-gray-800 mb-2">
          다음 기업을 삭제하려고 합니다:
        </p>
        <div className="text-base font-semibold text-red-600 mb-2">{companyName}</div>
        <p className="text-sm text-gray-700 mb-4">
          삭제 시 해당 기업의 계약금 수령 정보가 시스템에서 완전히 제거되며 복구할 수 없습니다.
        </p>

        <div className="bg-gray-50 border border-gray-200 rounded p-3 text-sm text-gray-800 mb-2 space-y-1">
          <p><strong>진행 상태:</strong> 송금 승인 완료</p>
        </div>
        <div className="bg-red-50 border border-red-300 text-red-800 rounded p-3 text-sm space-y-1 mb-4">
          <p><strong>삭제되는 항목:</strong> 기업 기본정보, 계약금액, 캐시백 금액, 송금 요청 내역, 승인/반려 이력</p>
        </div>

        

        <div className="flex justify-end space-x-2">
          <Button variant="outline" onClick={onClose}>취소</Button>
          <Button className="bg-red-600 text-white hover:bg-red-700" onClick={onConfirm}>
            삭제하기
          </Button>
        </div>
      </div>
    </div>
  );
}
