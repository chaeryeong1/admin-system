import { useState } from "react";
import { Card, CardContent } from "@/components/ui/card";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Table, TableHeader, TableRow, TableHead, TableBody, TableCell } from "@/components/ui/table";

export default function CashbackList() {
  const [filter, setFilter] = useState("");
  const [selectedId, setSelectedId] = useState(null);

  const [cashbacks, setCashbacks] = useState([
    {
      id: 1,
      company: "주식회사 이엑스",
      amount: 2000000,
      status: "요청대기",
      requestDate: "2025-04-10",
    },
  ]);

  const filtered = cashbacks.filter((item) => item.company.includes(filter));

  const view = (item) => {
    alert(`기업명: ${item.company}\n요청일: ${item.requestDate}\n금액: ${item.amount.toLocaleString()}원\n상태: ${item.status}`);
  };

  return (
    <div className="min-h-screen relative">
      <header className="bg-white border-b px-6 py-3 text-sm font-semibold flex space-x-6">
        <span className="text-gray-500">사업 관리</span>
        <span className="text-gray-500">기업 관리</span>
        <span className="text-gray-500">사업 안내</span>
        <span className="text-gray-500">신청</span>
        <span className="text-gray-500">선정 확인</span>
        <span className="text-gray-500">계약</span>
        <span className="text-gray-500">세금계산서 발행</span>
        <span className="text-gray-500">계약금 수령</span>
        <span className="text-blue-600">캐시백 송금</span>
        <span className="text-gray-500">장비 납품</span>
        <span className="text-gray-500">라이선스 발행</span>
        <span className="text-gray-500">SW 납품</span>
        <span className="text-gray-500">잔금 수령</span>
      </header>

      <div className="flex">
        <aside className="w-60 bg-gray-50 p-4 text-sm border-r h-full">
          <h2 className="text-base font-semibold mb-4">캐시백 송금</h2>
          <ul className="space-y-2">
            <li className="text-gray-600 hover:text-blue-600 cursor-pointer">ㆍ수령 기업 추가</li>
            <li className="text-blue-600 font-semibold">ㆍ송금 요청</li>
            <li className="text-gray-600 hover:text-blue-600 cursor-pointer">ㆍ송금 승인</li>
            <li className="text-gray-600 hover:text-blue-600 cursor-pointer">ㆍ송금</li>
          </ul>
        </aside>

        <div className="flex-1 p-6 space-y-6">
          <div className="text-xl font-bold border-b pb-2">캐시백 송금 관리 &gt; 송금 요청</div>

          <div className="flex items-center space-x-4">
            <Input
              placeholder="기업명 검색"
              value={filter}
              onChange={(e) => setFilter(e.target.value)}
              className="max-w-sm"
            />
            <Button variant="outline" onClick={() => {
              const match = cashbacks.find((c) => c.company.includes(filter));
              if (match) view(match);
              else alert("검색된 항목이 없습니다.");
            }}>조회</Button>
            <div className="flex-1 text-right">
              <Button onClick={() => alert('송금 요청 모달')}>송금 요청</Button>
            </div>
          </div>

          <Card>
            <CardContent className="p-4">
              <Table>
                <TableHeader>
                  <TableRow>
                    <TableHead>선택</TableHead>
                    <TableHead>기업명</TableHead>
                    <TableHead>요청일</TableHead>
                    <TableHead>금액</TableHead>
                    <TableHead>상태</TableHead>
                  </TableRow>
                </TableHeader>
                <TableBody>
                  {filtered.map((item) => (
                    <TableRow key={item.id}>
                      <TableCell>
                        <input
                          type="radio"
                          name="select-cashback"
                          checked={selectedId === item.id}
                          onChange={() => setSelectedId(item.id)}
                        />
                      </TableCell>
                      <TableCell>{item.company}</TableCell>
                      <TableCell>{item.requestDate}</TableCell>
                      <TableCell>{item.amount.toLocaleString()}원</TableCell>
                      <TableCell>{item.status === '요청대기' ? '요청대기' : '요청완료'}</TableCell>
                    </TableRow>
                  ))}
                </TableBody>
              </Table>
            </CardContent>
          </Card>

          <div className="flex justify-end space-x-4">
            <Button variant="destructive" onClick={() => {
              if (!selectedId) return alert("삭제할 항목을 선택하세요.");
              setCashbacks(cashbacks.filter((item) => item.id !== selectedId));
              setSelectedId(null);
            }}>삭제</Button>
          </div>
        </div>
      </div>
    </div>
  );
}
