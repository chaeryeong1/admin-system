import { useEffect } from "react";
import { Button } from "@/components/ui/button";

export default function CashbackDeleteModal({ receiver = {}, onClose, onDelete }) {
  useEffect(() => {
    if (!receiver || receiver.status === "승인") {
      alert("승인된 건은 삭제할 수 없습니다.");
      onClose();
    }
  }, [receiver, onClose]);

  const handleDelete = () => {
    if (receiver.status === "승인") {
      alert("승인된 건은 삭제할 수 없습니다.");
      return;
    }
    onDelete(receiver.id);
    onClose();
  };

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[420px]">
        <h2 className="text-lg font-semibold mb-4">🗑 송금 요청 삭제</h2>

        <div className="text-sm text-gray-800 mb-4 space-y-1">
          <p><strong>기업명:</strong> 주식회사 이엑스</p>
          <p><strong>대표자:</strong> 홍길동</p>
          <p><strong>계약금액:</strong> 2,000,000원</p>
        </div>

        <div className="bg-red-50 border border-red-300 text-red-700 text-sm rounded px-3 py-2 mb-4">
          ⚠️ 이 작업은 되돌릴 수 없습니다. 승인 전 상태에서만 삭제 가능합니다.
        </div>

        <div className="flex justify-end gap-2">
          <Button variant="outline" onClick={onClose}>취소</Button>
          <Button onClick={handleDelete} className="bg-red-600 text-white hover:bg-red-700">삭제 확정</Button>
        </div>
      </div>
    </div>
  );
}
