import { useEffect } from "react";
import { Button } from "@/components/ui/button";
import { RefreshCw } from "lucide-react";

export default function CashbackApprovalModal({ onClose, onConfirm, info }) {
  useEffect(() => {
    const handleEsc = (e) => {
      if (e.key === "Escape") onClose();
    };
    document.addEventListener("keydown", handleEsc);
    return () => document.removeEventListener("keydown", handleEsc);
  }, [onClose]);

  const mockInfo = {
    name: "주식회사 이엑스",
    ceo: "홍길동",
    contractAmount: 2000000,
    cashbackAmount: 200000,
    bankName: "국민은행",
    accountNumber: "110-123-456789",
    holder: "홍길동",
    attachmentName: "통장사본.pdf",
    note: "확인 부탁드립니다.",
  };

  const data = info || mockInfo;

  return (
    <div className="fixed inset-0 bg-black bg-opacity-40 flex items-center justify-center z-50">
      <div className="bg-white rounded-lg shadow-lg p-6 w-[460px]">
        <h2 className="text-lg font-semibold mb-4 flex items-center space-x-2">
          <RefreshCw className="text-blue-600" size={18} />
          <span>송금 재승인</span>
        </h2>

        <div className="text-sm text-yellow-800 bg-yellow-100 border border-yellow-300 rounded px-3 py-2 mb-4">
          ⚠️ 송금 전 상태일 때 승인 실수 건을 재승인할 수 있습니다.
        </div>

        <div className="text-sm text-gray-800 space-y-2">
          <p><strong>기업명:</strong> {data.name}</p>
          <p><strong>대표자:</strong> {data.ceo}</p>
          <p><strong>계약금액:</strong> {Number(data.contractAmount).toLocaleString()}원</p>
          <p><strong>캐시백 금액:</strong> {Number(data.cashbackAmount).toLocaleString()}원</p>
          <p><strong>계좌정보:</strong> {data.bankName} / {data.accountNumber} / {data.holder}</p>
          <p><strong>첨부 파일:</strong> {data.attachmentName ? <a href="#" className="text-blue-600 underline">{data.attachmentName}</a> : "없음"}</p>

          <div>
            <p className="text-sm font-medium">실무자 메모</p>
            <textarea
              className="mt-1 w-full border border-gray-300 rounded px-3 py-2 text-sm bg-gray-100"
              rows={3}
              readOnly
              value={data.note || "-"}
            />
          </div>

          <p className="text-sm font-medium">비고</p>
          <textarea
            className="mt-2 w-full border border-gray-300 rounded px-3 py-2 text-sm"
            rows={3}
            placeholder="예: 관리자가 승인한 정보를 보내드리니 위 내용으로 송금 부탁드립니다."
          />
        </div>

        <div className="flex justify-end space-x-2 mt-6">
          <Button className="text-gray-700 hover:text-black border border-gray-300" variant="ghost" onClick={onClose}>닫기</Button>
          <Button className="bg-[#0F172A] text-white hover:bg-[#1E293B]" onClick={onConfirm}>재승인</Button>
        </div>
      </div>
    </div>
  );
}
