import { useState } from "react";
import { Button } from "@/components/ui/button";
import { Input } from "@/components/ui/input";
import { Table, TableHeader, TableRow, TableHead, TableBody, TableCell } from "@/components/ui/table";

const initialApprovals = [
  {
    id: 1,
    name: "주식회사 이엑스",
    bizNo: "123-45-67890",
    ceo: "홍길동",
    contractAmount: 2000000,
    cashbackAmount: 200000,
    date: "2025-04-11",
    status: "대기"
  },
  {
    id: 2,
    name: "에버그로스",
    bizNo: "234-56-78901",
    ceo: "김예진",
    contractAmount: 1500000,
    cashbackAmount: 150000,
    date: "2025-04-12",
    status: "승인"
  }
];

export default function CashbackApprovalList() {
  const [companies, setCompanies] = useState(initialApprovals);
  const [filter, setFilter] = useState("");
  const [selectedId, setSelectedId] = useState(null);

  const filtered = companies.filter((c) => c.name.includes(filter));

  const getStatusBadge = (status) => {
    const base = "text-xs px-2 py-0.5 rounded border select-none";
    if (status === "송금완료") return <span className={`${base} bg-neutral-800 text-white border-neutral-700`}>송금완료</span>;
    if (status === "승인") return <span className={`${base} bg-green-50 text-green-700 border-green-400`}>승인</span>;
    return <span className={`${base} bg-gray-50 text-gray-400 border-gray-200`}>대기</span>;
  };

  return (
    <div className="p-6 space-y-4">
      <h2 className="text-xl font-bold pb-2 border-b">송금 승인</h2>

      <div className="flex gap-2 items-center">
        <select className="border rounded px-3 py-2 text-sm">
          <option>2025 클라우드 보급사업</option>
        </select>
        <Input
          placeholder="기업명 검색"
          value={filter}
          onChange={(e) => setFilter(e.target.value)}
          className="w-60"
        />
        <Button variant="outline">조회</Button>
        <div className="flex-1" />
      </div>

      <div className="overflow-hidden rounded-lg border border-gray-200 px-4 py-3">
        <Table>
          <TableHeader>
            <TableRow>
              <TableHead>선택</TableHead>
              <TableHead>기업명</TableHead>
              <TableHead>사업자등록번호</TableHead>
              <TableHead>대표자</TableHead>
              <TableHead>계약금액</TableHead>
              <TableHead>캐시백금액</TableHead>
              <TableHead>처리일</TableHead>
              <TableHead>상태</TableHead>
            </TableRow>
          </TableHeader>
          <TableBody>
            {filtered.map((c) => (
              <TableRow key={c.id}>
                <TableCell>
                  <label className="inline-flex items-center">
                    <input
                      type="radio"
                      name="select"
                      className="w-4 h-4 text-blue-600 bg-gray-100 border-gray-300 focus:ring-blue-500 focus:ring-2 rounded-full"
                      checked={selectedId === c.id}
                      onChange={() => setSelectedId(c.id)}
                    />
                  </label>
                </TableCell>
                <TableCell>
                  <a href="#" className="text-blue-600 underline text-sm">
                    {c.name}
                  </a>
                </TableCell>
                <TableCell>{c.bizNo}</TableCell>
                <TableCell>{c.ceo}</TableCell>
                <TableCell>{c.contractAmount.toLocaleString()}원</TableCell>
                <TableCell>{c.cashbackAmount.toLocaleString()}원</TableCell>
                <TableCell>{c.date}</TableCell>
                <TableCell>{getStatusBadge(c.status)}</TableCell>
              </TableRow>
            ))}
          </TableBody>
        </Table>
      </div>

      <div className="flex justify-end gap-2 pt-4">
        <Button variant="outline">승인</Button>
        <Button variant="destructive">반려</Button>
      </div>
    </div>
  );
}
